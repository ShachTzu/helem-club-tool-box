The `Lightbox` component displays a single gallery item (image or video) full-size inside an overlay, along with its title, artist name, tagged domains and previous/next navigation. It is RTL-first, matching the Helam Club community design language.

## Basic usage

Render a lightbox for a single gallery item, controlled by an `open` boolean:

```tsx
import React, { useState } from 'react';
import { Lightbox } from '@helemclub/gallery.ui.lightbox';

function BasicExample() {
  const [open, setOpen] = useState(true);

  const item = {
    id: `1`,
    slug: `quiet-after-the-storm`,
    title: `שקט אחרי הסערה`,
    description: `יצירה שמבטאת את הרוגע שמגיע אחרי גל של הצפה רגשית.`,
    mediaType: `image` as const,
    mediaUrl: `https://storage.googleapis.com/bit-generated-images/images/image_calm_abstract_art_piece_symbol_0_1785184760565.png`,
    artistName: `נועה ל.`,
    domains: [`ויסות רגשי`],
    createdAt: `2026-01-12T09:00:00.000Z`,
  };

  return (
    <div>
      <button onClick={() => setOpen(true)}>פתיחת יצירה</button>
      <Lightbox open={open} onClose={() => setOpen(false)} item={item} />
    </div>
  );
}
```

## Navigating between items

Provide `onPrev` / `onNext` callbacks along with `hasPrev` / `hasNext` flags to enable navigation arrows, and `currentIndex` / `totalCount` to show a position counter:

```tsx
import React, { useState } from 'react';
import { Lightbox } from '@helemclub/gallery.ui.lightbox';

function GalleryExample() {
  const [index, setIndex] = useState(0);
  const [open, setOpen] = useState(true);

  const items = [
    {
      id: `1`,
      slug: `quiet-after-the-storm`,
      title: `שקט אחרי הסערה`,
      mediaType: `image` as const,
      mediaUrl: `https://storage.googleapis.com/bit-generated-images/images/image_calm_abstract_art_piece_symbol_0_1785184760565.png`,
      artistName: `נועה ל.`,
      domains: [`ויסות רגשי`],
      createdAt: `2026-01-12T09:00:00.000Z`,
    },
    {
      id: `2`,
      slug: `light-at-the-edge`,
      title: `אור בקצה`,
      mediaType: `image` as const,
      mediaUrl: `https://storage.googleapis.com/bit-generated-images/images/image_serene_illustration_of_light_b_0_1785184760843.png`,
      artistName: `אנונימי`,
      domains: [`דיכאון ותחושת תקיעות`],
      createdAt: `2026-02-03T09:00:00.000Z`,
    },
  ];

  return (
    <div>
      <button onClick={() => setOpen(true)}>פתיחת גלריה</button>
      <Lightbox
        open={open}
        onClose={() => setOpen(false)}
        item={items[index]}
        hasPrev={index > 0}
        hasNext={index < items.length - 1}
        onPrev={() => setIndex((current) => Math.max(0, current - 1))}
        onNext={() => setIndex((current) => Math.min(items.length - 1, current + 1))}
        currentIndex={index + 1}
        totalCount={items.length}
      />
    </div>
  );
}
```

## Video items

The lightbox supports video media types, rendering a native `video` element with controls instead of an image:

```tsx
import React, { useState } from 'react';
import { Lightbox } from '@helemclub/gallery.ui.lightbox';

function VideoExample() {
  const [open, setOpen] = useState(true);

  const item = {
    id: `3`,
    slug: `breathe-and-begin`,
    title: `נשימה והתחלה`,
    description: `סרטון קצר של תרגיל נשימה מודרך בזריחה.`,
    mediaType: `video` as const,
    mediaUrl: `https://storage.googleapis.com/bit-generated-images/images/image_thumbnail_frame_for_a_short_mi_0_1785184760493.png`,
    artistName: `דנה כ.`,
    domains: [`מיינדפולנס ונשימות`, `חרדה`],
    createdAt: `2026-03-01T09:00:00.000Z`,
  };

  return (
    <div>
      <button onClick={() => setOpen(true)}>צפייה בסרטון</button>
      <Lightbox open={open} onClose={() => setOpen(false)} item={item} />
    </div>
  );
}
```
